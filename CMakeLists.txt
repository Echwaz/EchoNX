cmake_minimum_required(VERSION 3.16)
set(CMAKE_SYSTEM_NAME Generic)
set(CMAKE_C_COMPILER /opt/devkitpro/devkitA64/bin/aarch64-none-elf-gcc)
set(CMAKE_CXX_COMPILER /opt/devkitpro/devkitA64/bin/aarch64-none-elf-g++)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
project(echonx VERSION 0.4.6 LANGUAGES C CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DEPS REQUIRED sdl2 SDL2_ttf SDL2_image libavformat libavcodec libswresample libavutil libcurl)
add_executable(echonx src/app.cpp src/api.cpp src/player.cpp)
set_target_properties(echonx PROPERTIES SUFFIX ".elf" CXX_STANDARD 17)
target_compile_definitions(echonx PRIVATE __SWITCH__)
target_include_directories(echonx PRIVATE include vendor /opt/devkitpro/libnx/include ${DEPS_STATIC_INCLUDE_DIRS})
target_compile_options(echonx PRIVATE -march=armv8-a+crc+crypto -mtune=cortex-a57 -mtp=soft -fPIE -ffunction-sections -fdata-sections ${DEPS_STATIC_CFLAGS_OTHER})
target_link_directories(echonx PRIVATE /opt/devkitpro/libnx/lib ${DEPS_STATIC_LIBRARY_DIRS})
target_link_options(echonx PRIVATE -specs=/opt/devkitpro/libnx/switch.specs -Wl,--gc-sections)
target_link_libraries(echonx PRIVATE ${DEPS_STATIC_LIBRARIES} nx mbedtls mbedx509 mbedcrypto)
add_custom_target(echonx.nro ALL DEPENDS echonx
    COMMAND /opt/devkitpro/tools/bin/nacptool --create "EchoNX" "Community prototype" "${PROJECT_VERSION}" echonx.nacp
    COMMAND /opt/devkitpro/tools/bin/elf2nro echonx.elf echonx.nro --icon=${CMAKE_SOURCE_DIR}/icon.jpg --nacp=echonx.nacp --romfsdir=${CMAKE_SOURCE_DIR}/romfs)
