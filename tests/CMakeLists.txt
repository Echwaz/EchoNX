cmake_minimum_required(VERSION 3.16)
project(echonx_tests LANGUAGES CXX)
enable_testing()
find_package(SDL2 REQUIRED)

add_executable(listening_history_test listening_history_test.cpp)
add_executable(local_state_test local_state_test.cpp)
add_executable(album_art_mode_test album_art_mode_test.cpp)
add_executable(seek_bar_test seek_bar_test.cpp)
add_executable(multi_artist_test multi_artist_test.cpp)
add_executable(artwork_plan_test artwork_plan_test.cpp)
add_executable(artwork_cache_test artwork_cache_test.cpp)
add_executable(disk_cache_test disk_cache_test.cpp)
add_executable(queue_actions_test queue_actions_test.cpp)
add_executable(playback_order_test playback_order_test.cpp)
add_executable(album_cover_test album_cover_test.cpp)
add_executable(library_store_test library_store_test.cpp)
add_executable(visuals_test visuals_test.cpp)
foreach(test IN ITEMS listening_history_test local_state_test album_art_mode_test library_store_test visuals_test album_cover_test playback_order_test queue_actions_test disk_cache_test artwork_cache_test artwork_plan_test multi_artist_test seek_bar_test)
    target_compile_features(${test} PRIVATE cxx_std_17)
    target_compile_definitions(${test} PRIVATE SDL_MAIN_HANDLED)
    target_include_directories(${test} PRIVATE ../include ../vendor)
    add_test(NAME ${test} COMMAND ${test})
endforeach()
target_link_libraries(visuals_test PRIVATE SDL2::SDL2)
set_tests_properties(visuals_test PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")

target_link_libraries(album_art_mode_test PRIVATE SDL2::SDL2)
set_tests_properties(album_art_mode_test PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")

find_package(Threads REQUIRED)
target_link_libraries(local_state_test PRIVATE Threads::Threads)

find_package(PkgConfig REQUIRED)
pkg_check_modules(UI REQUIRED IMPORTED_TARGET SDL2_ttf SDL2_image)
add_executable(app_regression_test app_regression_test.cpp ../src/player_preview.cpp)
target_compile_features(app_regression_test PRIVATE cxx_std_17)
target_compile_definitions(app_regression_test PRIVATE SDL_MAIN_HANDLED ECHO_PREVIEW ECHO_TEST)
target_include_directories(app_regression_test PRIVATE ../include ../vendor)
target_link_libraries(app_regression_test PRIVATE SDL2::SDL2 PkgConfig::UI Threads::Threads)
add_test(NAME app_regression_test COMMAND app_regression_test)
set_tests_properties(app_regression_test PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy;ECHONX_STATE_DIR=${CMAKE_CURRENT_BINARY_DIR}/app-state" WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")

find_file(TEST_FONT NAMES DejaVuSans.ttf arial.ttf PATHS /usr/share/fonts/truetype/dejavu C:/Windows/Fonts)
if(TEST_FONT)
    set_property(TEST app_regression_test APPEND PROPERTY ENVIRONMENT "ECHONX_FONT=${TEST_FONT}" "ECHONX_CLOCK_FONT=${CMAKE_CURRENT_SOURCE_DIR}/../romfs/fonts/Barlow-SemiBold.ttf")
endif()

pkg_check_modules(AUDIO QUIET IMPORTED_TARGET libavformat libavcodec libswresample libavutil)
find_program(FFMPEG ffmpeg)
if(AUDIO_FOUND AND FFMPEG)
    add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/tone.m4a COMMAND ${FFMPEG} -loglevel error -y -f lavfi -i sine=frequency=440:duration=3 -c:a aac ${CMAKE_CURRENT_BINARY_DIR}/tone.m4a)
    add_custom_target(audio_fixture DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/tone.m4a)
    add_executable(player_regression_test player_regression_test.cpp ../src/player.cpp)
    target_compile_features(player_regression_test PRIVATE cxx_std_17)
    target_compile_definitions(player_regression_test PRIVATE SDL_MAIN_HANDLED)
    target_include_directories(player_regression_test PRIVATE ../include ../vendor)
    target_link_libraries(player_regression_test PRIVATE SDL2::SDL2 PkgConfig::AUDIO Threads::Threads)
    add_dependencies(player_regression_test audio_fixture)
    add_test(NAME player_regression_test COMMAND player_regression_test ${CMAKE_CURRENT_BINARY_DIR}/tone.m4a)
    set_tests_properties(player_regression_test PROPERTIES ENVIRONMENT "SDL_AUDIODRIVER=dummy")
endif()

pkg_check_modules(CURL QUIET IMPORTED_TARGET libcurl)
if(UNIX AND CURL_FOUND)
