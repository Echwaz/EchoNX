cmake_minimum_required(VERSION 3.16)
project(echonx_tests LANGUAGES CXX)
enable_testing()
find_package(SDL2 REQUIRED)

add_executable(queue_actions_test queue_actions_test.cpp)
add_executable(playback_order_test playback_order_test.cpp)
add_executable(album_cover_test album_cover_test.cpp)
add_executable(library_store_test library_store_test.cpp)
add_executable(visuals_test visuals_test.cpp)
foreach(test IN ITEMS library_store_test visuals_test album_cover_test playback_order_test queue_actions_test)
    target_compile_features(${test} PRIVATE cxx_std_17)
    target_compile_definitions(${test} PRIVATE SDL_MAIN_HANDLED)
    target_include_directories(${test} PRIVATE ../include ../vendor)
    add_test(NAME ${test} COMMAND ${test})
endforeach()
target_link_libraries(visuals_test PRIVATE SDL2::SDL2)
set_tests_properties(visuals_test PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
